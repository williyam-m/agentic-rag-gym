Quick Start

# 1. get your API token
$ curl -X POST /api/v1/auth/login -d '{"username":"you","password":"pw"}'
# 2. create a domain
$ curl -X POST /api/v1/rag/domains -H "Authorization: Bearer $TOKEN" \
  -d '{"name":"my-project","description":"My RAG domain"}'
# 3. upload documents
$ curl -X POST /api/v1/rag/domains/$ID/documents -F "file=@data.txt"
# 4. generate rewards
$ curl -X POST /api/v1/rewards/compute \
  -d '{"algorithm":"grpo","query":"...","answer":"...","retrieved_docs":[...]}'
